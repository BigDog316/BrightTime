with matched as (
  select ls.listing_id, ls.site_id, ls.slug, ls.title, ls.summary, ls.org_id, ls.kind,
         ls.cost_cents, ls.is_free, ls.modality, ls.neighbourhood_id,
         ls.next_start_at, ls.next_deadline_at, ls.popularity, ls.published_at, ls.is_drop_in,
         st_distance(ls.location, st_setsrid(st_makepoint(@origin_lon, @origin_lat), 4326)::geography) as distance_m,
         ts_rank_cd(ls.search_vector, websearch_to_tsquery(@ts_config::regconfig, @text), 32) as text_rank,
         0 as interest_overlap
  from opportunity.listing_search ls
  where ls.tenant_id = @tenant_id and ls.is_published
    and ls.kind = any(@kinds)
    and ls.facets @> @facets_all
    and ls.session_window && tstzrange(@window_start, @window_end)
    and ls.cost_cents <= @max_cost_cents
    and ls.location && st_makeenvelope(@box_west, @box_south, @box_east, @box_north, 4326)::geography
    and st_distance(ls.location, st_setsrid(st_makepoint(@origin_lon, @origin_lat), 4326)::geography) <= @radius_m
    and (ls.search_vector @@ websearch_to_tsquery(@ts_config::regconfig, @text)
         or @text <<% ls.fuzzy_text)
),
best_site as (
  select distinct on (listing_id) *
  from matched
  order by listing_id, distance_m asc nulls last
),
scored as (
  select bs.*, round((0.45 * least(text_rank * 4.0, 1.0) + 0.20 * (case when distance_m is null then 0.5 else exp(-distance_m / 4000.0) end) + 0.15 * (case when next_start_at is null then 0.3 else exp(-greatest(extract(epoch from (next_start_at - @as_of)) / 86400.0, 0) / 30.0) end) + 0.12 * least(interest_overlap / 3.0, 1.0) + 0.08 * least(popularity / 100.0, 1.0))::numeric, 6) as score
  from best_site bs
)
select s.listing_id, s.site_id, s.slug, s.title, s.summary, s.org_id, s.kind,
       s.cost_cents, s.is_free, s.modality, s.neighbourhood_id,
       s.next_start_at, s.next_deadline_at, s.published_at,
       s.distance_m, s.score, s.interest_overlap, s.text_rank, s.popularity,
       extract(epoch from (s.next_start_at - @as_of)) / 86400.0 as days_until_start
from scored s
order by s.score desc, s.listing_id desc
limit @page_size
